RNA.InTargetClassKLeTwo {n : ℕ} (T : SecondaryStructure n) : Prop
@[expose] def RNA.InTargetClassKLeTwo : {n : ℕ} → SecondaryStructure n → Prop :=
fun {n} T =>
  shortHelixCount T ≤ 2 ∧
    (∀ (H : MaximalHelix T), H.length ≠ 1) ∧
      (∀ (H : MaximalHelix T), H.length ≠ 2 → 3 ≤ H.length) ∧ ¬HasM5 T ∧ ¬HasM3Dot T
