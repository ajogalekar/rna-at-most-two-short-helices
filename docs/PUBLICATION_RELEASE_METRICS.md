### Metric method

- Physical lines are Python `str.splitlines()` counts over UTF-8 source.  The
  aggregate values were independently cross-checked with `wc -l` and agree.
- Code lines are physical lines with non-whitespace remaining after a lexical
  state machine removes `--` comments and nested `/- ... -/` comments.  The
  scanner retains quoted string contents, so comment-like text in strings is
  not removed.
- A declaration is counted when the first residual token on a physical line,
  after any same-line `@[...]` attributes and any of the modifiers `private`,
  `protected`, `noncomputable`, `partial`, `public`, or `local`, is one of the
  command keywords in the declaration table.  These are reproducible
  syntactic source-command counts: compiler-generated auxiliaries are not
  counted, and multiple constants generated by one source command still count
  as one command.
- The authoritative closure manifest is
  `/Users/ashujo/Documents/Science/rna_at_most_two_release_qualification/isolated_run/final_dependency_closure.txt`.
  It contains exactly 46 nonempty, unique paths; every path exists and is
  tracked.  Its SHA-256 is
  `4dda2cb3d78cec0bb6c5dd34237abe05ec0a35903a9e4beb24979ff4d3256d34`.

The following scanner reproduces the aggregate metrics above when run from
the repository root:

```python
from collections import Counter
from pathlib import Path
import re, subprocess

ROOT = Path.cwd()
MANIFEST = Path(
    "/Users/ashujo/Documents/Science/rna_at_most_two_release_qualification/"
    "isolated_run/final_dependency_closure.txt"
)
KEYWORDS = ("def", "abbrev", "structure", "inductive", "theorem", "lemma",
            "class", "instance", "opaque", "axiom", "example")
MODIFIERS = {"private", "protected", "noncomputable", "partial", "public", "local"}

def git_lines(*args):
    p = subprocess.run(["git", *args], check=True, text=True,
                       stdout=subprocess.PIPE)
    return [line for line in p.stdout.splitlines() if line]

def strip_comments(text):
    out, line = [], []
    i = block_depth = 0
    in_line_comment = in_string = escaped = False
    while i < len(text):
        ch, nxt = text[i], text[i:i + 2]
        if ch == "\n":
            out.append("".join(line)); line = []
            in_line_comment = escaped = False
            i += 1; continue
        if in_line_comment:
            i += 1; continue
        if block_depth:
            if nxt == "/-": block_depth += 1; i += 2
            elif nxt == "-/": block_depth -= 1; i += 2
            else: i += 1
            continue
        if in_string:
            line.append(ch)
            if escaped: escaped = False
            elif ch == "\\": escaped = True
            elif ch == '"': in_string = False
            i += 1; continue
        if nxt == "--": in_line_comment = True; i += 2
        elif nxt == "/-": block_depth = 1; i += 2
        else:
            line.append(ch)
            if ch == '"': in_string = True
            i += 1
    if line or (text and not text.endswith("\n")): out.append("".join(line))
    return out

def after_attributes(s):
    s = s.lstrip()
    while s.startswith("@["):
        depth = 0; in_string = escaped = False; end = None
        for i, ch in enumerate(s[1:], 1):
            if in_string:
                if escaped: escaped = False
                elif ch == "\\": escaped = True
                elif ch == '"': in_string = False
            elif ch == '"': in_string = True
            elif ch == "[": depth += 1
            elif ch == "]":
                depth -= 1
                if depth == 0: end = i + 1; break
        if end is None: return s
        s = s[end:].lstrip()
    return s

def metrics(paths):
    total = Counter()
    for rel in paths:
        text = (ROOT / rel).read_text(encoding="utf-8")
        lines = strip_comments(text)
        count = Counter(physical=len(text.splitlines()),
                        code=sum(bool(line.strip()) for line in lines))
        for line in lines:
            s = after_attributes(line)
            while True:
                m = re.match(r"([A-Za-z_][A-Za-z0-9_']*)\b", s)
                if not m or m.group(1) not in MODIFIERS: break
                s = s[m.end():].lstrip()
            m = re.match(r"([A-Za-z_][A-Za-z0-9_']*)\b", s)
            if m and m.group(1) in KEYWORDS:
                count[m.group(1)] += 1; count["declarations"] += 1
        total.update(count)
    total["files"] = len(paths)
    total["comment_or_blank"] = total["physical"] - total["code"]
    return total

tracked = git_lines("ls-files", "*.lean")
closure = [line for line in MANIFEST.read_text().splitlines() if line]
publication = ["RNA/AtMostTwoShort/PublicationExamples.lean",
               "RNA/AtMostTwoShort/PublicationExamplesAxiomAudit.lean"]
for name, paths in (("tracked", tracked), ("closure", closure),
                    ("publication", publication),
                    ("prospective_union", tracked + publication)):
    print(name, metrics(paths))
```
